\section{I.\ Motivation \& Research Question}

\begin{frame}{Sharpe's Arithmetic \& Gruber's Puzzle}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{The tautology (Sharpe 1991)}
      \begin{itemize}\itemsep2pt
        \item Before costs, the average active dollar \emph{must} match the market.
        \item After costs, it \emph{must} underperform.
        \item \alert{Not an empirical claim — it is arithmetic.}
      \end{itemize}
      \vspace{4pt}
      \textbf{Three decades of evidence}
      \begin{itemize}\itemsep2pt
        \item Jensen (1968), Carhart (1997), FF (2010): active funds deliver negative net alpha on average.
        \item BSW (2010): fraction of genuinely skilled managers is negligible.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Yet the puzzle persists (Gruber 1996)}
      \begin{itemize}\itemsep2pt
        \item US active equity funds hold $>$\$10 trillion in AUM.
        \item Capital inflows are structurally positive.
        \item Flow–performance convexity (ST 1998) is intact.
      \end{itemize}
      \vspace{4pt}
      \begin{block}{The Active Management Puzzle}
        \small Why do investors persistently allocate to
        strategies that underperform passive alternatives net of fees?
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Two Competing Explanations}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Rational Equilibrium}
        \textbf{Berk \& Green (2004)}
        \begin{itemize}\itemsep2pt
          \item Skilled managers exist; rational investors chase them.
          \item Capital inflows erode skill advantages (decreasing returns to scale).
          \item Equilibrium: net alpha $\approx 0$ in value-weighted terms.
          \item Puzzle is an artefact of equal-weighting.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Behavioral Frictions}
        \textbf{Four channels tested here}
        \begin{itemize}\itemsep2pt
          \item \textbf{H1} Sentiment amplifies winner-chasing.
          \item \textbf{H2} Loss aversion locks in losers.
          \item \textbf{H3} Lottery appeal weakens exit discipline.
          \item \textbf{H4} Fee sensitivity relaxes under optimism.
        \end{itemize}
        \footnotesize If operative, flows contain a behavioral component
        beyond rational updating.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{III.\ Aggregate Performance}

\begin{frame}{Carhart (1997) Four-Factor Model}
  \[
    R_{i,t} - R_{f,t} \;=\; \alpha_i + \beta_i \,\text{MKT}_t
                            + s_i \,\text{SMB}_t
                            + h_i \,\text{HML}_t
                            + p_i \,\text{MOM}_t
                            + \varepsilon_{i,t}
  \]
  \begin{itemize}\itemsep5pt
    \item Time-series regression; alpha annualised ($\times 12$).
    \item \textbf{Newey–West HAC} standard errors (6-month lag) throughout.
    \item Portfolio-level estimation (EW and VW aggregate portfolio series).
    \item Net returns: gross return less $\text{ER}/12$ each month.
  \end{itemize}
\end{frame}

\section{IV.\ Bootstrap \& FDR Decomposition}

\begin{frame}{Why Bootstrap? The Multiple-Testing Problem}
  \begin{itemize}\itemsep5pt
    \item With $N \approx 1{,}800$ funds, some will show extreme $t(\hat{\alpha})$
          \emph{by luck alone} — even if every fund's true alpha is zero.
    \item OLS $p$-values are fund-by-fund and ignore cross-sectional correlation.
    \item \textbf{Fama \& French (2010)} solution: joint calendar-month resampling
          preserves the cross-sectional correlation structure.
    \item Generate a ``zero-skill'' benchmark distribution for the \emph{cross-section}
          of $t$-statistics.
    \item \textbf{BSW (2010)} complement: use the marginal $p$-value distribution to
          decompose the cross-section into skilled / unskilled / zero-alpha funds.
  \end{itemize}
\end{frame}

\begin{frame}{FF (2010) Bootstrap: Design}
  \textbf{Steps}
  \begin{enumerate}\itemsep3pt
    \item Estimate $\hat{\alpha}_i$ for each fund; compute zero-alpha returns
          $\tilde{r}_{i,t} = r_{i,t} - \hat{\alpha}_i$.
    \item Draw $B = 10{,}000$ bootstrap samples by resampling \emph{calendar months}
          (same months for all funds — preserves cross-section).
    \item Re-estimate Carhart on each bootstrapped series; record $t(\hat{\alpha}^b)$.
    \item Compare \emph{actual} cross-sectional $t$-percentiles to the simulated distribution.
  \end{enumerate}
  \vspace{6pt}
\end{frame}

\begin{frame}{BSW (2010) Decomposition: Methodology}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \small\textbf{Step 1 — Estimate $\hat{\pi}_0$}\\[3pt]
      Using Storey (2002) on Newey--West $p$-values:
      \[
        \hat{\pi}_0 = \frac{|\{p_i > \lambda\}|}{N(1-\lambda)}, \quad \lambda = 0.5
      \]
      \footnotesize Funds with $p_i > 0.5$ are unlikely to have non-zero true alpha;
      their density provides a conservative upper bound on the zero-alpha mass.
    \end{column}
    \begin{column}{0.47\textwidth}
      \small\textbf{Step 2 — Decompose tails}\\[3pt]
      \footnotesize At significance level $\gamma$:
      \begin{itemize}\itemsep2pt
        \item $S^-_\gamma$, $S^+_\gamma$: observed fractions with sig.\ negative / positive $t(\hat{\alpha})$.
        \item $F_\gamma = \hat{\pi}_0 \cdot \gamma/2$: false discoveries per tail.
        \item $\hat{\pi}^-_A = S^-_\gamma - F_\gamma$: genuinely unskilled.
        \item $\hat{\pi}^+_A = S^+_\gamma - F_\gamma$: genuinely skilled.
      \end{itemize}
      \vspace{3pt}
      \footnotesize Reference row: $\gamma = 0.20$.\\
      $\hat{\pi}^+_A < 0$: right-tail significance does not exceed false-discovery rate.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Performance Persistence: KTWW Bootstrap Design}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small\textbf{Kosowski, Timmermann, Wermers \& White (2006)}
      \begin{itemize}\itemsep2pt
        \item \textbf{Formation}: rank active funds into deciles D1--D10
              by Carhart $t(\hat{\alpha})$.
              D1 = best past performers; D10 = worst.
        \item \textbf{Holding period}: EW decile portfolios; holding-period
              Carhart alpha pooled across non-overlapping cohorts
              (36-month formation, 12-month holding).
        \item \textbf{Bootstrap}: resample residuals $B=10{,}000$ under
              zero-persistence null. $p^R_B$ ($p^L_B$): fraction of
              simulated $t(\hat{\alpha})$ exceeding (falling below)
              the observed holding-period $t$-statistic.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Why bootstrap over OLS?}
        \footnotesize
        \begin{itemize}\itemsep1pt
          \item OLS $p^C$ compares against the standard normal —
                the null for a \emph{randomly drawn} fund.
          \item D1 and D10 are not random draws: they were
                \emph{selected because they were extreme}. Even under
                zero skill, a selected extreme decile has had a luck run
                that spills into the holding period. Its null has
                \emph{fatter tails} — $p^C$ overstates significance.
          \item The bootstrap re-runs the full select-then-test procedure
                $B=10{,}000$ times under zero skill, generating the correct
                fat-tailed null. $p^L_B$ is the valid inferential basis.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{IX.\ H1: Sentiment-Convexity}

\begin{frame}{H1: Hypothesis \& Prediction}
  \begin{block}{H1: Sentiment-Convexity}
    During high-sentiment regimes, the winner-segment flow response steepens
    ($\delta_3 > 0$), amplifying flow–performance convexity.
  \end{block}
  \vspace{4pt}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \small\textbf{Behavioral mechanism}
      \begin{itemize}\itemsep3pt
        \item Representativeness heuristic (Kahneman \& Tversky 1974): investors
              extrapolate recent winner returns as evidence of skill.
        \item Sentiment amplifies this extrapolation.
        \item Prediction: $\delta_3 > 0$ (convexity amplified at the top).
        \item Secondary: $\delta_3 - \delta_1 > 0$ (asymmetry widens).
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \small\textbf{Rational null}
      \begin{itemize}\itemsep3pt
        \item BG: sentiment should not alter the flow–performance mapping.
        \item Search-cost story (ST 1998): marketing, not sentiment.
        \item Prediction: $\delta_1 = \delta_2 = \delta_3 = 0$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

    + \theta\, X_{i,t-1}
    + \mu_i + \varepsilon_{i,t}

    \begin{column}{0.50\textwidth}

\section{X.\ H2: Disposition Effect / Illusion of Control}

\begin{frame}{H2: Hypothesis \& Discrimination Design}
  \begin{block}{H2: Disposition Effect / Illusion of Control}
    \small Do behavioral constraints prevent investors from exiting underperformers,
    or does financial stress \emph{accelerate} their exit? H2 discriminates between
    two channels with \emph{opposing} predictions for the loser-segment interaction $\delta_1$.
  \end{block}
  \vspace{2pt}
  \small\textbf{Two channels, opposing predictions for $\delta_1$:}
  \vspace{2pt}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Behavioral retention} ($\delta_1 < 0$, given $\beta_1 > 0$)
      \begin{itemize}\itemsep1pt
        \item \textit{Disposition effect} (Shefrin \& Statman 1985):
              loss aversion prevents realising losses — loser-segment slope attenuated.
        \item \textit{Illusion of control} (Barber \& Odean 2001):
              overconfident belief in market predictability sustains retention of losers.
        \item Proxy: Call-to-put ratio ($D^{\text{INV-PCR}}$) — high C/P reflects
              directional overconfidence.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Funding constraints} ($\delta_1 > 0$, given $\beta_1 > 0$)
      \begin{itemize}\itemsep1pt
        \item \textit{Brunnermeier \& Pedersen (2009)}: margin pressure forces
              liquidation — loser exit \emph{accelerates}.
        \item Loser-segment slope \emph{steepens}, not flattens.
        \item Proxy: Detrended margin debt ($D^{\text{MD,Det}}$).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{H2: Regression Specifications}
  \footnotesize\textbf{Standard spec} (Cols.\ 2--3): piecewise model with financial-stress state $Z_t$:
  \vspace{-2pt}
  \[
    \text{Flow}_{i,t+1} = \sum_q \beta_q R^q_{i,t}
    + \sum_q \delta_q (R^q_{i,t}\times Z_t)
    + \gamma Z_t + \theta\,X_{i,t-1} + \mu_i + \varepsilon_{i,t}
  \]
  \vspace{-2pt}
  \begin{tabular}{@{}ll@{}}
(2): & $Z_t = D^{\text{MD,Det}}_t$ — top-34\% indicator for $\Delta^{yoy}(\text{MD}/\text{MCAP})$ (year-over-year change in margin debt ratio) \\
    (3): & $Z_t = D^{\text{INV-PCR}}_t$ — top-34\% indicator for CBOE equity call-put ratio \\
  \end{tabular}\\[5pt]
  \footnotesize\textbf{Discriminant spec} (Col.\ 4): adds $D^{\text{SENT}}_t$ alongside $D^{\text{MD,Det}}_t$ with full rank interactions to test channel separation:
  \vspace{-2pt}
  \[
    \text{Flow}_{i,t+1} = \sum_q \beta_q R^q_{i,t}
    + \sum_q \delta^{\text{MD}}_q (R^q_{i,t}\times D^{\text{MD,Det}}_t)
    + \sum_q \delta^{\text{S}}_q (R^q_{i,t}\times D^{\text{SENT}}_t)
    + \gamma^{\text{MD}} D^{\text{MD,Det}}_t + \gamma^{\text{S}} D^{\text{SENT}}_t
    + \theta\,\mathbf{X}_{i,t-1} + \mu_i + \varepsilon_{i,t}
  \]
  \vspace{-2pt}
  \footnotesize\textbf{Controls} $\mathbf{X}_{i,t-1}$: $\log(\text{TNA})$, $\log(\text{Age})$, Return volatility, Style flow.\\
  \vspace{4pt}
  Fund FE; Ticker $\times$ yearmo clustering. $N=232{,}800$ (Cols.\ 2,4); $168{,}488$ (Col.\ 3, PCR sample 2003--2019).\\
  \vspace{4pt}
  \textbf{Test}: joint $\chi^2_3$ ($H_0: \delta_1=\delta_2=\delta_3=0$); headline: $\delta_1 > 0$ one-sided (loser-end steepening).
\end{frame}

  \begin{itemize}\itemsep3pt
    \item \textbf{Baseline $\beta_1 = 0.031^{***}$} — loser-segment slope is positive, confirming the sign framework: $\delta_1 > 0$ steepens exit, $\delta_1 < 0$ would attenuate it.
    \item \textbf{$\delta_1 > 0$} with high one-sided $p$: rejects disposition/illusion-of-control ($\delta_1 < 0$) decisively.
    \item High margin debt \emph{steepens} the loser segment — losers bleed \emph{more} under stress.
    \item \textbf{Brunnermeier–Pedersen (2009)} signature: liquidity spiral, not loss aversion.
    \item INV-PCR: clean null — illusion of control not operative in this sample.

  \end{itemize}
\end{frame}

      \begin{itemize}\itemsep3pt

\begin{frame}{The Post-2011 Regime: Three Competing Readings}
  \begin{enumerate}\itemsep8pt

          efficiency improvements. $\hat{\pi}^-_A$ rose to 35.4\% in P3.

          active-fund investors toward the most behavioral tail, amplifying the
          cross-sectional underperformance of small, lottery-like funds.

\begin{frame}{A1: Sharpe's Arithmetic — Market Decomposition}

  \textbf{Setup.} The market is divided into two mutually exclusive and exhaustive segments:
  \[
    V_M = V_P + V_A
  \]
  where $V_M$ is total market value, $V_P$ is total value held by passive investors,
  and $V_A$ is total value held by active investors.\\[6pt]

  \textbf{Value-weight shares:}
  \[
    w_P = \frac{V_P}{V_M} \qquad w_A = \frac{V_A}{V_M} \qquad w_P + w_A = 1
  \]

  \textbf{Market return identity.} The market return is the value-weighted average
  of passive and active returns:
  \[
    R_M = w_P \cdot R_P + w_A \cdot R_A
  \]

  \textbf{Passive return.} By construction, passive investors hold the market portfolio:
  \[
    R_P = R_M
  \]

\end{frame}

\begin{frame}{A2: Sharpe's Arithmetic — The Identity}

  \textbf{Derivation.} Substitute $R_P = R_M$ into the market return equation:
  \[
    R_M = w_P \cdot R_M + w_A \cdot R_A
  \]
  Rearrange:
  \[
    R_M (1 - w_P) = w_A \cdot R_A
    \qquad\Longrightarrow\qquad
    R_M \cdot w_A = w_A \cdot R_A
  \]
  Divide both sides by $w_A > 0$:
  \[
    \boxed{R_A = R_M}
  \]
  \alert{Before costs, the aggregate gross return of active management equals the market return.}

  \vspace{6pt}
  \textbf{Net-of-cost corollary.} Let $c_A > c_P \geq 0$ be per-period cost rates:
  \[
    NR_A - NR_P = (R_M - c_A) - (R_M - c_P) = c_P - c_A < 0
  \]
  \[
    \boxed{NR_A < NR_P}
  \]
  \alert{After costs, the average active investor must underperform the average passive investor.
  Always. In every period.}\\[4pt]

\end{frame}

\begin{frame}{A3: What the Identity Does — and Does Not — Imply}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small\textbf{What it \emph{does} establish}
      \begin{itemize}\itemsep2pt
        \item A \emph{tautological} result: follows from the definition
              of value-weighted returns and market clearing alone.
        \item The average active dollar underperforms by exactly
              $c_A - c_P$ per period.
        \item Holds for \emph{any} period, asset class, or market structure.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small\textbf{What it \emph{does not} establish}
      \begin{itemize}\itemsep2pt
        \item Nothing about the \emph{distribution}: some managers
              outperform; others underperform by more to compensate.
        \item Nothing about \emph{investor} returns: skilled fund
              selection could allow outperformance net of fees.
        \item Nothing about \emph{persistence}: an empirical question.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{5pt}
  \begin{block}{The Bridge to This Dissertation}
    \small The identity explains why the \emph{average} active fund underperforms.
    The puzzle is why investors \emph{persist} in allocating to the below-average
    tail — which is where H1–H4 live.
    The identity is the starting point, not the answer.
  \end{block}
\end{frame}

      \small\textbf{Raw data \& scaling}
      \begin{itemize}\itemsep2pt
        \item Source: FINRA monthly debit balances in margin accounts (Rule 4521(d)).
              Available from January 1997; H2 window January 1998--December 2023.
        \item Scale by Wilshire 5000 Full Cap Index (.W5KF, LSEG):
              \[
                \text{MDRatio}(t) = \frac{\text{MarginDebt}(t)}{\text{W5KF}(t)}
              \]
        \item Year-over-year differencing removes seasonality:
              \[
                \Delta^{yoy}\text{MDRatio}(t) = \frac{\text{MDRatio}(t) - \text{MDRatio}(t-12)}{\text{MDRatio}(t-12)}
              \]
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \small\textbf{Regime dummy}
      \begin{itemize}\itemsep2pt
        \item Following Baker--Wurgler (2007): top-34\% indicator
              \[
                D^{\text{MD,Det}}_t = \mathbf{1}\!\left[\Delta^{yoy}\text{MDRatio}(t) > Q_{66}\right]
              \]
        \item $Q_{66}$ fixed on full 1998--2023 sample; no look-ahead bias.
      \end{itemize}
      \vspace{3pt}
      \small\textbf{Robustness proxy}
      \begin{itemize}\itemsep2pt
        \item CBOE equity call-to-put ratio ($D^{\text{INV-PCR}}$): bottom-34\% indicator for PCR — high C/P reflects illusion of control.
        \item Available 1995--2023; Col (3) of H2 table.
      \end{itemize}
    \end{column}
  \end{columns}
